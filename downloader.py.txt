logger = logging.getLogger("downloader")
failure_logger = logging.getLogger("failed")

def save_file(temp_path, dest_path, url, post_time, creator_config):
    dest_folder = os.path.dirname(dest_path)
    os.makedirs(dest_folder, exist_ok=True)

    unpacked_files = []
    if creator_config['AUTO_UNZIP'] and is_archive(dest_path):
        logger.info(f"Extracting... -> {dest_path}")

        try:
            unpacked_files = unpack(temp_path, creator_config['ALLOWED_EXTENSIONS'], creator_config['ZIP_PASSWORDS'])
            
            if len(unpacked_files) > 0:
                zipfile_folder = os.path.splitext(dest_path)[0]
                os.makedirs(zipfile_folder, exist_ok=True)
                
                index = 0
                for file_path in unpacked_files:
                    filename = f"{os.path.basename(os.path.splitext(file_path[0])[0])}_{file_path[1]}{os.path.splitext(file_path[0])[1]}"
                    file_dest_path = f"{zipfile_folder}/{os.path.basename(zipfile_folder).split('_')[0]}_{filename}"
                    shutil.move(file_path, file_dest_path)
                    os.utime(file_dest_path, (post_time, post_time))

                    unpacked_files[index] = file_dest_path
                    index += 1
                
                os.utime(zipfile_folder, (post_time, post_time))

        except Exception as e:
            logger.error(f"Failed extracting (password may be required) -> {dest_path}")
            logger.debug(f"[Exception] {e}")

            if not creator_config['KEEP_FAILED_UNZIPS']:
                os.remove(temp_path)
                return False

    shutil.move(temp_path, dest_path)
    os.utime(dest_path, (post_time, post_time))
    os.utime(dest_folder, (post_time, post_time))

    with get_hashes_lock():
        hashes_path = get_hashes_filepath(post_folder)
        hashes = load_hashes(hashes_path)
        hashes[generate_hash(dest_path)] = {
            'url': url,
            'filepath': dest_path,
            'unpacked_files': unpacked_files
        }
        save_hashes(hashes, hashes_path)
    
def download_file(file: tuple[str], post_folder: str, post_time: float, creator_config: dict, max_attempts: int = 60) -> bool:
    url = file[0]
    filename = file[1]

    temp_folder = '/temp'
    temp_filename_base = f"{post_folder.split('/')[0].split('_')[-1]}_{post_folder.split('/')[1].split('_')[-1]}"
    temp_path = os.path.join(temp_folder, f"{temp_filename_base}_{filename}")

    dest_folder = '/data/' + post_folder
    dest_path = os.path.join(dest_folder, filename)

    logger.debug(f"Download starting... -> {dest_path}")
    exception = None
    for attempt in range(max_attempts):
        try:
            r = SESSION.get(url.split('f=')[0], headers=HEADERS, stream=True, timeout=3600) # old: 
            r.raise_for_status()

            with open(temp_path, 'wb') as f:
                for chunk in r.iter_content(8192):
                    if chunk:
                        f.write(chunk)
            
            break
        
        except Exception as e:
            wait_time = 1 #2**attempt

            if isinstance(e, HTTPError) and e.response.status_code == 502:
                wait_time = 5

                #logger.debug(f"Cannot connect to server -> 502 Bad Gateway")
                #logger.debug(f"[Exception] {e}")
                #exception = e
                #break

            logger.warning(f"Attempt failed ({attempt+1}/{max_attempts}) -> {dest_path}")
            logger.debug(f"[Exception] {e}")

            if attempt < max_attempts-1:
                logger.debug(f"Starting again in {wait_time}s... -> {dest_path}")
                time.sleep(wait_time)
            else:
                exception = e

    if exception:
        if os.path.exists(temp_path):
            os.remove(temp_path)

        logger.error(f"Download failed -> {dest_path}")

        failure = {
            "file": temp_path,
            "dest": dest_path,
            "url": url,
            "error": str(exception),
            "attempts": max_attempts,
            "timestamp": datetime.datetime.now().isoformat() + "Z"
        }
        failure_logger.debug(json.dumps(failure, ensure_ascii=False))

        return False

    unpacked_files = []

    if creator_config['AUTO_UNZIP'] and is_zipfile(filename):
        logger.info(f"Extracting... -> {dest_path}")

        try:
            unpacked_files = unpack(temp_path, creator_config['ALLOWED_EXTENSIONS'], creator_config['ZIP_PASSWORDS'])
            
            if len(unpacked_files) > 0:
                zipfile_folder = os.path.splitext(dest_path)[0]
                os.makedirs(zipfile_folder, exist_ok=True)
                
                index = 0
                for file_path in unpacked_files:
                    file_dest_path = f"{zipfile_folder}/{os.path.basename(zipfile_folder).split('_')[0]}_{os.path.basename(file_path)}"
                    shutil.move(file_path, file_dest_path)
                    os.utime(file_dest_path, (post_time, post_time))

                    unpacked_files[index] = file_dest_path
                    index += 1
                
                os.utime(zipfile_folder, (post_time, post_time))

        except RuntimeError as e:
            logger.error(f"Failed extraction ({e}) -> {dest_path}")
            logger.debug(f"[Exception] {e}")

            if not creator_config['KEEP_FAILED_UNZIPS']:
                os.remove(temp_path)
                return False
            
        except Exception as e:
            logger.error(f"Failed extraction -> {dest_path}")
            logger.debug(f"[Exception] {e}")

            if not creator_config['KEEP_FAILED_UNZIPS']:
                os.remove(temp_path)
                return False

    os.makedirs(dest_folder, exist_ok=True)
    shutil.move(temp_path, dest_path)
    os.utime(dest_path, (post_time, post_time))
    os.utime(dest_folder, (post_time, post_time))

    with get_hashes_lock():
        hashes_path = get_hashes_filepath(post_folder)
        hashes = load_hashes(hashes_path)
        hashes[generate_hash(dest_path)] = {
            'url': url,
            'filepath': dest_path,
            'unpacked_files': unpacked_files
        }
        save_hashes(hashes, hashes_path)
    
    logger.info(f"Download successful -> {dest_path}")
    return True

def concurrent_download(files: list[tuple], creator_config: dict, max_workers: int = 5) -> dict[tuple[str, str], bool]:
    results = {}
    with ThreadPoolExecutor(max_workers=max_workers) as exe:
        futures = {exe.submit(download_file, file, folder, post_time, creator_config): file for file, folder, post_time in files}

        for fut in as_completed(futures):
            file = futures[fut]
            results[file] = fut.result()

    return results

def download_creator(creator: dict):
    creator_name = sanitize_filename(creator['name'])
    creator_id = creator['id']
    creator_service = creator['service']

    logger.info(f"\n\n----------|| {creator_name} - {creator_service} - {creator_id} ||----------\n")
    
    logger.info(f"Loading config...")
    creator_config = load_creator_config(creator_id)

    logger.info(f"Fetching creator posts...")
    posts = get_all_posts_from_creator(creator_service, creator_id)

    logger.info(f"Found {len(posts)} posts")
    logger.info("Detecting files in posts...")

    all_files = []
    skipped = 0

    for post in posts:
        post_files = detect_files_in_post(post, creator_config['ALLOWED_EXTENSIONS'], creator_config['DOWNLOAD_THUMBNAILS'], creator_config['DOWNLOAD_ATTACHMENTS'], creator_config['DOWNLOAD_EMBEDDED_IMAGES'])

        post_time = get_post_time(post['published'])
        post_title = sanitize_filename(post['title'])
        post_id = post['id']

        post_folder = f"{creator_name}_{creator_service}_{creator_id}/{post_title}_{post_id}"

        hashes = load_hashes(get_hashes_filepath(post_folder))

        for file in post_files:
            if get_hash_from_url(file[0]) in hashes:
                skipped += 1
                continue

            all_files.append((file, post_folder, post_time))

    logger.info(f"Found {len(all_files)} files ({skipped} skipped)")
    if len(all_files) == 0:
        logger.info("Skipping...")
        time.sleep(3)
        return

    logger.info("Starting download...")

    res = concurrent_download(all_files, creator_config)

    success = sum(v for v in res.values())
    logger.info(f"Downloaded {success}/{len(all_files)} files")
    time.sleep(5)
